# Enter your code here. Read input from STDIN. Print output to STDOUT
from collections import deque

n, m = map(int, input().split())

grid = []
q = deque()
fresh = 0

for i in range(n):
    row = list(map(int, input().split()))
    grid.append(row)

    for j in range(m):
        if row[j] == 2:
            q.append((i, j))
        elif row[j] == 1:
            fresh += 1

time = 0

directions = [
    (-1, 0),  # up
    (1, 0),   # down
    (0, -1),  # left
    (0, 1)    # right
]

while q and fresh > 0:
    size = len(q)
    time += 1

    for _ in range(size):
        r, c = q.popleft()

        for dr, dc in directions:
            nr = r + dr
            nc = c + dc

            if 0 <= nr < n and 0 <= nc < m and grid[nr][nc] == 1:
                grid[nr][nc] = 2
                fresh -= 1
                q.append((nr, nc))

if fresh == 0:
    print(time)
else:
    print(-1)
