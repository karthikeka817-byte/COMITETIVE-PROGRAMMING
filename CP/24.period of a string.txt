# Enter your code here. Read input from STDIN. Print output to STDOUT
S = input().strip()

n = len(S)
lps = [0] * n

j = 0
for i in range(1, n):
    while j > 0 and S[i] != S[j]:
        j = lps[j - 1]

    if S[i] == S[j]:
        j += 1

    lps[i] = j

period = n - lps[-1]

if n % period == 0:
    print(period)
else:
    print(n)
