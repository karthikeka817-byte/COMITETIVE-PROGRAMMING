# Enter your code here. Read input from STDIN. Print output to STDOUT
S = input().strip()

seen = 0

# First pass: track which characters occur
for ch in S:
    bit = 1 << (ord(ch) - ord('a'))
    seen |= bit

# Find duplicates and print them in first-occurrence order
result = []
printed = 0
counts = 0

for ch in S:
    bit = 1 << (ord(ch) - ord('a'))

    if counts & bit:
        # Already counted once before
        continue

    # Check if this character appears more than once
    first_pos = S.find(ch)
    second_pos = S.find(ch, first_pos + 1)

    if second_pos != -1:
        result.append(ch)
        counts |= bit

if result:
    print(*result)
else:
    print("No duplicates")
