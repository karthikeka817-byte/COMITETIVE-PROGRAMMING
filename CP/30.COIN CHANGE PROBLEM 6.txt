import sys

data = list(map(int, sys.stdin.read().split()))

V = data[0]
coins = data[1:]

INF = V + 1
dp = [INF] * (V + 1)

dp[0] = 0

for amount in range(1, V + 1):
    for coin in coins:
        if coin <= amount:
            dp[amount] = min(dp[amount], dp[amount - coin] + 1)

if dp[V] == INF:
    print(-1)
else:
    print(dp[V])
