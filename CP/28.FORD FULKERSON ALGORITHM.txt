from collections import deque

V, E = map(int, input().split())

capacity = [[0] * V for _ in range(V)]

for _ in range(E):
    u, v, c = map(int, input().split())
    capacity[u][v] += c

max_flow = 0

while True:
    parent = [-1] * V
    parent[0] = 0

    q = deque([0])

    while q and parent[V - 1] == -1:
        u = q.popleft()

        for v in range(V):
            if parent[v] == -1 and capacity[u][v] > 0:
                parent[v] = u
                q.append(v)

                if v == V - 1:
                    break


    if parent[V - 1] == -1:
        break

    path_flow = float('inf')
    v = V - 1

    while v != 0:
        u = parent[v]
        path_flow = min(path_flow, capacity[u][v])
        v = u

    v = V - 1

    while v != 0:
        u = parent[v]
        capacity[u][v] -= path_flow
        capacity[v][u] += path_flow
        v = u

    max_flow += path_flow

print(max_flow)
